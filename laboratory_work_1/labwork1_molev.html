def load_mnist_images(filename):
    """Загрузка изображения из MNIST файла"""
    with gzip.open(filename, 'rb') as f:
        magic, num, rows, cols = struct.unpack('>IIII', f.read(16))
        images = np.frombuffer(f.read(), dtype=np.uint8).reshape(num, rows * cols)
        return images / 255.0

def load_mnist_labels(filename):
    """Загрузка метки из MNIST файла"""
    with gzip.open(filename, 'rb') as f:
        magic, num = struct.unpack('>II', f.read(8))
        labels = np.frombuffer(f.read(), dtype=np.uint8)
        return labels

def relu(x):
        """Функция активации ReLU."""
        return np.maximum(0, x)
    
def relu_derivative(x):
        """Производная функции ReLU."""
        return (x > 0).astype(np.float32)
    
def softmax(x):
        """Функция активации softmax."""
        exp_x = np.exp(x)
        return exp_x / np.sum(exp_x, axis=1, keepdims=True)
    
def compute_loss(y_true, y_pred):
        """Вычисление кросс-энтропийной ошибки для одного примера."""
        m = y_true.shape[0]
        loss = -np.log(y_pred[np.arange(m), np.argmax(y_true, axis=1)] + 1e-8)
        return np.sum(loss) / m

class TwoLayerNeuralNetworkSGD:
    """Двухслойная полносвязная нейронная сеть с одним скрытым слоем."""
    
    def __init__(self, input_size, hidden_size, output_size, learning_rate=0.1):
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.learning_rate = learning_rate
        
        # Инициализация весов и смещений
        self.W1 = np.random.randn(input_size, hidden_size) * np.sqrt(2.0 / input_size)
        self.b1 = np.zeros((1, hidden_size))
        
        self.W2 = np.random.randn(hidden_size, output_size) * np.sqrt(2.0 / hidden_size)
        self.b2 = np.zeros((1, output_size))
        
    def forward(self, X):
        """Прямое распространение."""
        # Скрытый слой
        self.hidden_input = np.dot(X, self.W1) + self.b1
        self.hidden_output = relu(self.hidden_input)
        
        # Выходной слой
        self.output_input = np.dot(self.hidden_output, self.W2) + self.b2
        self.output = softmax(self.output_input)
        
        return self.output
    
    def backward(self, X, y_true, y_pred):
        """Обратное распространение ошибки."""
        
        m = X.shape[0]
        # Градиент по выходному слою
        dL2 = y_pred.copy()
        dL2[np.arange(m), np.argmax(y_true, axis=1)] -= 1
        dL2 /= m
        
        # Градиенты для W2 и b2
        dW2 = np.dot(self.hidden_output.T, dL2)
        db2 = np.sum(dL2, axis=0, keepdims=True)
        
        # Градиент по скрытому слою
        dA1 = np.dot(dL2, self.W2.T)
        dL1 = dA1 * relu_derivative(self.hidden_input)
        
        # Градиенты для W1 и b1
        dW1 = np.dot(X.T, dL1)
        db1 = np.sum(dL1, axis=0, keepdims=True)
        
        # Обновление параметров
        self.W2 -= self.learning_rate * dW2
        self.b2 -= self.learning_rate * db2
        self.W1 -= self.learning_rate * dW1
        self.b1 -= self.learning_rate * db1
    
    def predict(self, X):
        """Предсказание меток для входных данных."""
        output = self.forward(X)
        return np.argmax(output, axis=1)
    
    def accuracy(self, X, y_true):
        """Вычисление точности классификации"""
        predictions = self.predict(X)
        true_labels = np.argmax(y_true, axis=1)
        return np.mean(predictions == true_labels)
    
    def shuffle_data(self, X, y):
        """Перемешивание данных"""
        indices = np.random.permutation(X.shape[0])
        return X[indices], y[indices]
    
    def train(self, X_train, y_train, epochs, batch_size):
        """Обучение"""
        print("Начало обучения:")
        
        for epoch in range(epochs):
            start_time = time.time()
            epoch_loss = 0
            correct_predictions = 0
            
            # Перемешиваем данные
            X_shuffled, y_shuffled = self.shuffle_data(X_train, y_train)
            
            # обучение
            for i in range(0, len(X_train), batch_size):

                X_batch = X_shuffled[i:i + batch_size]
                y_batch = y_shuffled[i:i + batch_size]
                
                # Прямой и обратный проход
                output = self.forward(X_batch)
                self.backward(X_batch, y_batch, output)
            
            # Метрики после эпохи
            loss = compute_loss(y_train, self.forward(X_train))
            accuracy = self.accuracy(X_train, y_train)
            epoch_time = time.time() - start_time
            
            print(f"Эпоха {epoch+1:2d}/{epochs} | "
                  f"Время: {epoch_time:5.2f}с | "
                  f"Ошибка: {loss:.4f} | "
                  f"Точность: {accuracy:.4f}")
    
    def test(self, X_test, y_test):
        """Тестирование нейронной сети"""
        # Прямой проход для всех тестовых данных
        output = self.forward(X_test)
        
        # Вычисление ошибки и точности
        loss = compute_loss(y_test, output)
        accuracy = self.accuracy(X_test, y_test)
        
        print(f"\nРезультаты на тестовой выборке:")
        print(f"   Точность: {accuracy:.4f}")
        print(f"   Ошибка: {loss:.4f}")
        
        return accuracy, loss
